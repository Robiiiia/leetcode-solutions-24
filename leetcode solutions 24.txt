190. reverse bits



public class Solution {

    public int reverseBits(int n) {

        int result = 0; // Здесь будем собирать число с обратным порядком битов

        for (int i = 0; i < 32; i++) { // Обрабатываем ровно 32 бита исходного числа

            result = result << 1; // Сдвигаем результат влево, освобождая место для нового бита

            result = result | (n & 1); // Извлекаем последний бит n с помощью & 1 и добавляем его в result через OR (|)

            n = n >>> 1; // Сдвигаем n вправо на 1 бит, чтобы следующий бит оказался последним
        }

        return result; // Возвращаем число с развёрнутыми 32 битами
    }
}
